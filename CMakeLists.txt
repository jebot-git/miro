cmake_minimum_required(VERSION 3.22)
project(miro-pdf VERSION 0.12.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET fontconfig zlib gio-2.0)
find_package(OpenGL REQUIRED)
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_program(CARGO cargo REQUIRED)
find_program(MIRO_UNZIP unzip REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_program(MIRO_CC cc REQUIRED)
execute_process(COMMAND ${MIRO_CC} -print-file-name=include OUTPUT_VARIABLE GCC_INCLUDES OUTPUT_STRIP_TRAILING_WHITESPACE)
set(MIRO_BINDGEN_ARGS "-isystem ${GCC_INCLUDES}" CACHE STRING "Additional bindgen compiler arguments")
set(MIRO_CARGO_HOME "$ENV{CARGO_HOME}" CACHE PATH "Cargo registry and toolchain cache")
if(NOT MIRO_CARGO_HOME)
  set(MIRO_CARGO_HOME "$ENV{HOME}/.cargo")
endif()
set(MIRO_CARGO_TARGET "${CMAKE_BINARY_DIR}/cargo" CACHE PATH "Rust build directory")
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS src/*.rs)
add_custom_command(OUTPUT "${MIRO_CARGO_TARGET}/release/libmiro_backend.a"
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_HOME=${MIRO_CARGO_HOME}" "CARGO_TARGET_DIR=${MIRO_CARGO_TARGET}" "BINDGEN_EXTRA_CLANG_ARGS=${MIRO_BINDGEN_ARGS}" ${CARGO} check --locked --release --lib
  COMMAND ${CMAKE_COMMAND} -E env "CARGO_HOME=${MIRO_CARGO_HOME}" "CARGO_TARGET_DIR=${MIRO_CARGO_TARGET}" "BINDGEN_EXTRA_CLANG_ARGS=${MIRO_BINDGEN_ARGS}" ${CARGO} build --locked --release --lib -j2
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  DEPENDS ${CORE_SOURCES} Cargo.toml Cargo.lock assets/default.conf
  VERBATIM USES_TERMINAL)
add_custom_target(miro-core DEPENDS "${MIRO_CARGO_TARGET}/release/libmiro_backend.a")
add_executable(miro-pdf fltk/main.cpp fltk/worker.cpp fltk/viewer.cpp fltk/rpc.cpp)
add_dependencies(miro-pdf miro-core)
target_compile_features(miro-pdf PRIVATE cxx_std_17)
target_compile_options(miro-pdf PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(miro-pdf PRIVATE "${MIRO_CARGO_TARGET}/release/libmiro_backend.a" fltk::gl-shared fltk::images-shared fltk::fltk-shared OpenGL::GL PkgConfig::NATIVE nlohmann_json::nlohmann_json Threads::Threads PNG::PNG ${CMAKE_DL_LIBS} m)
# MuPDF bundles a private FreeType without bitmap drivers. Keep its archive
# symbols private: exporting FT_* interposes FLTK/Fontconfig's system FreeType
# and breaks raw BDF metadata, glyphs and live 2x enlargement.
target_link_options(miro-pdf PRIVATE "LINKER:--exclude-libs,ALL")
include(fltk/desktop-menu.cmake)
de_xr_desktop_menu(miro-pdf)
get_target_property(FLTK_LIBRARY fltk::fltk-shared IMPORTED_LOCATION_RELEASE)
get_filename_component(FLTK_LIBRARY_DIR "${FLTK_LIBRARY}" DIRECTORY)
set_target_properties(miro-pdf PROPERTIES BUILD_RPATH "${FLTK_LIBRARY_DIR}" INSTALL_RPATH "${FLTK_LIBRARY_DIR};$ORIGIN/../lib")
install(TARGETS miro-pdf RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/fonts/unifont-17.0.04.bdf DESTINATION ${CMAKE_INSTALL_DATADIR}/fonts/miro-pdf)
install(FILES assets/fonts/COPYING.unifont LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/miro-pdf)
install(FILES fltk/miro-pdf.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/logo.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps RENAME miro-pdf.png)
install(FILES assets/default.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/miro-pdf)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME document-core COMMAND ${CMAKE_COMMAND} -E env "CARGO_HOME=${MIRO_CARGO_HOME}" "CARGO_TARGET_DIR=${MIRO_CARGO_TARGET}" "BINDGEN_EXTRA_CLANG_ARGS=${MIRO_BINDGEN_ARGS}" ${CARGO} test --locked --release --lib)
  set_tests_properties(document-core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
